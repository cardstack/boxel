/* =============================================================================
 * Helpers — pure, no side effects, safe to call with undefined.
 * ============================================================================= */

function fmtMMSS(seconds: number | null | undefined): string {
  if (seconds == null || !Number.isFinite(seconds)) return '—';
  let s = Math.max(0, Math.floor(seconds));
  let m = Math.floor(s / 60);
  let r = s % 60;
  return `${m.toString().padStart(2, '0')}:${r.toString().padStart(2, '0')}`;
}

function fmtShortMMSS(seconds: number | null | undefined): string {
  // Same as fmtMMSS but drops leading zero on the minutes part when < 10.
  // Used for narrow timeline segments where every pixel counts.
  if (seconds == null || !Number.isFinite(seconds)) return '—';
  let s = Math.max(0, Math.floor(seconds));
  let m = Math.floor(s / 60);
  let r = s % 60;
  return `${m}:${r.toString().padStart(2, '0')}`;
}

function fmtNumber(n: number | null | undefined): string {
  if (n == null || !Number.isFinite(n)) return '—';
  return n.toLocaleString('en-US');
}

function fmtPercent(part: number, whole: number, decimals = 1): string {
  if (!whole || !Number.isFinite(whole) || !Number.isFinite(part)) return '—';
  return `${((part / whole) * 100).toFixed(decimals)}%`;
}

function fmtHitRate(fresh: number, cache: number): string {
  let total = (fresh || 0) + (cache || 0);
  if (total === 0) return '—';
  return `${((cache / total) * 100).toFixed(3)}%`;
}

function capitalise(s: string | null | undefined): string {
  if (!s) return '';
  return s.charAt(0).toUpperCase() + s.slice(1);
}

/* Map turnType → the CSS class suffix used to color it. */
function turnColorClass(turnType: string | null | undefined): string {
  switch (turnType) {
    case 'bootstrap':
      return 'bootstrap';
    case 'design':

      return 'machinery';

}

interface DerivedTurn {
  turn: TurnField;
  colorClass: string;
  isLive: boolean;
  durationSeconds: number; // 0 for live turn (elapsed used instead)
  durationDisplay: string;
  narrowLabel: boolean; // true → print duration BELOW the segment instead of inside
}

/* One arc of a KPI donut. The circle is drawn with r = 15.9155 so its
   circumference is ~100 — dash lengths are then percentages directly, no
   per-slice trig. `offset` starts each arc where the previous one ended,
   with 25 as the 12-o'clock origin. */
interface DonutSlice {
  label: string;
  valueDisplay: string;
  pctDisplay: string;
  colorClass: string;
  dash: string;
  offset: string;
}

function buildDonut(
  parts: { label: string; value: number; colorClass: string; valueDisplay: string }[],
): DonutSlice[] {
  let usable = parts.filter((p) => Number.isFinite(p.value) && p.value > 0);
  let total = usable.reduce((sum, p) => sum + p.value, 0);
  if (total <= 0) return [];
  let cumulative = 0;
  return usable.map((p) => {
    let pct = (p.value / total) * 100;
    let slice: DonutSlice = {
      label: p.label,
      valueDisplay: p.valueDisplay,
      pctDisplay: `${Math.round(pct)}%`,
      colorClass: p.colorClass,
      dash: `${pct.toFixed(2)} ${(100 - pct).toFixed(2)}`,
      offset: `${(125 - cumulative).toFixed(2)}`,
    };
    cumulative += pct;
    return slice;
  });
}

/* Ticket colors cycle through the data hues so each ticket owns one color
   across the timeline, its legend entry, and the inference donut. */
function ticketColorClass(index: number): string {
  return `t${index % 5}`;
}

/* Ticket status → the pill's wording and color. The tracker's vocabulary
   (backlog / in_progress / blocked / review / done) plus `running`, which
   telemetry sets when a turn is in flight regardless of what the index
   still says. An unknown or missing status renders no pill at all rather
   than inventing one. */
function statusPill(
  status: string | null | undefined,
): { label: string; className: string } | undefined {
  switch (status) {
    case 'running':
      return { label: 'Running', className: 'rt-status--running' };
    case 'in_progress':
      return { label: 'In progress', className: 'rt-status--progress' };
    case 'done':
      return { label: 'Completed', className: 'rt-status--done' };
    case 'backlog':
      return { label: 'Backlog', className: 'rt-status--backlog' };
    case 'blocked':
      return { label: 'Blocked', className: 'rt-status--blocked' };
    case 'review':
      return { label: 'In review', className: 'rt-status--review' };
    default:
      return undefined;
  }
}

/* How a skill reached the model: a fresh session pays to send it, a
   forked/resumed one already carries it in its cached prefix. */
function deliveryLabel(delivery: string | null | undefined): string {
  switch (delivery) {
    case 'inherited':
      return 'Already in context';
    case 'mixed':
      return 'Loaded, then inherited';
    default:
      return 'Loaded into agent';
  }
}

/* =============================================================================
 * Isolated component — the big one.
 * ============================================================================= */

class RunTelemetryIsolated extends Component<typeof RunTelemetry> {
  @tracked nowMs = Date.now();
  #ticker: ReturnType<typeof setInterval> | undefined;

  constructor(owner: unknown, args: any) {
    super(owner as any, args);
    // Minimal client-side clock: one setInterval that updates a single
    // tracked property. Only the TOTAL KPI number and the live timeline
    // segment width consume this; the rest of the template stays stable.
    this.#ticker = setInterval(() => {
      this.nowMs = Date.now();
    }, 1000);
    registerDestructor(this, () => {
      if (this.#ticker) clearInterval(this.#ticker);
    });
  }

  get config(): RunConfigField | undefined {
    return this.args.model?.config as RunConfigField | undefined;
  }
  get totals(): TotalsField | undefined {
    return this.args.model?.totals as TotalsField | undefined;
  }
  get turns(): TurnField[] {
    return (this.args.model?.turns as TurnField[] | undefined) ?? [];
  }
  get phases(): PhaseField[] {
    return (this.args.model?.phases as PhaseField[] | undefined) ?? [];
  }
  get events(): EventField[] {
    return (this.args.model?.events as EventField[] | undefined) ?? [];
  }
  get issues(): IssueSummaryField[] {
    return (this.args.model?.issues as IssueSummaryField[] | undefined) ?? [];
  }
  get skills(): SkillContextField[] {
    return (this.args.model?.skills as SkillContextField[] | undefined) ?? [];
  }
  get board(): BoardIssueField[] {
    return (this.args.model?.board as BoardIssueField[] | undefined) ?? [];
  }

  /* Tickets on the board the run has not taken a turn on yet — they own no
     turns, so nothing else in the card would mention them. */
  get notStartedTickets() {
    return this.board
      .filter((b) => !b.started)
      .map((b) => ({
        issueId: b.issueId ?? '',
        title: b.title ?? '',
        pill: statusPill(b.status),
      }));
  }

  get hasNotStartedTickets(): boolean {
    return this.notStartedTickets.length > 0;
  }

  get outcome(): string {
    return this.config?.outcome ?? '';
  }
  get isRunning(): boolean {
    return this.outcome === 'running';
  }

  /* --------------------------- Clock / durations ------------------------- */

  get startedAtMs(): number {
    let s = this.config?.startedAt;
    if (!s) return 0;
    let d = new Date(s as unknown as string).getTime();
    return Number.isFinite(d) ? d : 0;
  }

  get elapsedSeconds(): number {
    if (!this.startedAtMs) return 0;
    return Math.max(0, Math.floor((this.nowMs - this.startedAtMs) / 1000));
  }

  /* Wall-clock displayed by the TOTAL KPI:
   *   - finished / stopped / failed: use totals.wallClockSeconds
   *   - running:                     tick from Date.now() - startedAt
   */
  get wallClockDisplaySeconds(): number {
    if (this.isRunning) {
      return Math.max(this.elapsedSeconds, this.totals?.wallClockSeconds ?? 0);
    }
    return this.totals?.wallClockSeconds ?? 0;
  }
  get wallClockDisplay(): string {
    return fmtMMSS(this.wallClockDisplaySeconds);
  }

  /* Timeline uses per-turn duration; live turn uses elapsed since its startedAt. */
  turnElapsedSeconds(turn: TurnField): number {
    let s = turn?.startedAt;
    if (!s) return 0;
    let started = new Date(s as unknown as string).getTime();
    if (!Number.isFinite(started)) return 0;
    return Math.max(0, Math.floor((this.nowMs - started) / 1000));
  }

  turnWeight(turn: TurnField): number {
    let d = turn?.durationSeconds;
    if (d == null) return this.turnElapsedSeconds(turn);
    return d;
  }

  /* --------------------------- Header --------------------------------- */

  get runIndex(): string {
    // Best-effort: parse a numeric suffix out of the target-realm URL
    // slug (e.g. "run-telemetry-v4" → "4"), else fall back to "1".
    let url = this.config?.targetRealmUrl ?? '';
    let m = /v(\d+)/i.exec(url as unknown as string);
    if (m) return m[1];
    return '1';
  }
  get eyebrowText(): string {
    return `Run ${this.runIndex} · ${capitalise(this.config?.targetPhase)}`;
  }

  get headerTitle(): string {
    let m = this.args.model as any;
    let name = m?.cardInfo?.name;
    if (name) return name as string;
    if (m?.title) return m.title as string;
    return 'Untitled Run Telemetry';
  }

  get subtitleParts() {
    // Return a small structure the template can render as three inline
    // pieces so the middle piece can be <strong>.
    let issues = this.issues.length;
    let turns = this.turns.length;
    if (this.isRunning) {
      return {
        before: `${issues} issue${issues === 1 ? '' : 's'} started, ${turns} agent turn${turns === 1 ? '' : 's'} so far, `,
        strong: this.wallClockDisplay,
        after: ' elapsed on the wall clock.',
      };
    }
    // Only tickets that actually ran a validation pipeline have a verdict;
    // reading "every issue completed" off a default would be a fabrication.
    let validated = this.issues.filter((i) => (i.validationStepsRun ?? 0) > 0);
    let after: string;
    if (validated.length === 0) {
      after = ' — no validation pipeline ran.';
    } else if (validated.every((i) => i.validationPassed)) {
      after = ` with all ${validated.length} validated ticket${validated.length === 1 ? '' : 's'} passing.`;
    } else {
      after = ' with mixed validation outcomes.';
    }
    return {
      before: `${issues} issue${issues === 1 ? '' : 's'}, ${turns} agent turn${turns === 1 ? '' : 's'}, done in `,
      strong: fmtMMSS(this.totals?.wallClockSeconds),
      after,
    };
  }

  get startedAtDisplay(): string {
    let s = this.config?.startedAt as unknown as string | undefined;
    if (!s) return '—';
    try {
      let d = new Date(s);
      return d.toLocaleString('en-US', {
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
        day: '2-digit',
        month: 'short',
        year: 'numeric',
        hour12: false,
      });
    } catch {
      return '—';
    }
  }
  get endedAtDisplay(): string {
    let s = this.config?.endedAt as unknown as string | undefined;
    if (!s) return '—';
    try {
      let d = new Date(s);
      return d.toLocaleString('en-US', {
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
        day: '2-digit',
        month: 'short',
        year: 'numeric',
        hour12: false,
      });
    } catch {
      return '—';
    }
  }

  /* --------------------------- KPI row --------------------------------- */

  get inferenceDisplay(): string {
    return fmtMMSS(this.totals?.inferenceSeconds);
  }
  get inferencePercent(): string {
    let wall = this.wallClockDisplaySeconds;
    let inf = this.totals?.inferenceSeconds ?? 0;
    if (!wall) return '—';
    return `${Math.round((inf / wall) * 100)}%`;
  }
  get inferenceFootText(): string {
    return `${this.inferencePercent} of wall clock · ${this.turns.length} turn${this.turns.length === 1 ? '' : 's'}`;
  }

  get machinerySeconds(): number {
    let t = this.totals;
    if (!t) return 0;
    return (
      (t.validationSeconds ?? 0) +
      (t.renderGateSeconds ?? 0) +
      (t.syncSeconds ?? 0) +
      (t.startupSeconds ?? 0)
    );
  }
  get machineryDisplay(): string {
    return fmtMMSS(this.machinerySeconds);
  }
  /* Short by design: the tile's donut key already itemizes validation /
     render / sync / startup, and repeating it here made the tile so tall
     that the row could not hold four columns. */
  get machineryFootText(): string {
    let t = this.totals;
    if (!t) return '';
    let syncCalls = t.syncCallCount ?? 0;
    let wall = this.wallClockDisplaySeconds;
    let share = wall ? `${Math.round((this.machinerySeconds / wall) * 100)}%` : '—';
    return `${share} of wall clock · ${syncCalls} sync${syncCalls === 1 ? '' : 's'}`;
  }

  get compactionCount(): number {
    return this.turns.reduce((n, t) => n + (t.compacted ? 1 : 0), 0);
  }
  get compactionFootText(): string {
    let n = this.compactionCount;
    if (n === 0) return 'no context-window overflows';
    return `${n} × ~17k fresh tokens re-ingested`;
  }

  /* --------------------------- KPI donuts ------------------------------ */

  /* Wall clock split by where it went. The turn in flight gets its own
     slice while the run is live — it is not "unaccounted", it is the thing
     currently happening, and its elapsed time is the largest slice on a
     mid-run card. What remains genuinely is unaccounted: orchestration
     between spans (index settles, issue reads, run-log writes) that no
     span covers, and folding it into the others would overstate them. */
  get totalDonut(): DonutSlice[] {
    let wall = this.wallClockDisplaySeconds;
    let inference = this.totals?.inferenceSeconds ?? 0;
    let machinery = this.machinerySeconds;
    let live = this.issueGroupsForTimeline.reduce(
      (sum, g) => sum + g.liveSeconds,
      0,
    );
    let rest = Math.max(0, wall - inference - machinery - live);
    return buildDonut([
      {
        label: 'Model inference',
        value: inference,
        colorClass: 'inference',
        valueDisplay: fmtMMSS(inference),
      },
      {
        label: 'Turn in flight',
        value: live,
        colorClass: 'live',
        valueDisplay: fmtMMSS(live),
      },
      {
        label: 'Machinery',
        value: machinery,
        colorClass: 'machinery',
        valueDisplay: fmtMMSS(machinery),
      },
      {
        label: 'Unaccounted',
        value: rest,
        colorClass: 'unaccounted',
        valueDisplay: fmtMMSS(rest),
      },
    ]);
  }

  /* Inference split by ticket — the colors the timeline uses. Settled turns
     only, so the slices add up to the number printed on the tile. */
  get inferenceDonut(): DonutSlice[] {
    return buildDonut(
      this.issueGroupsForTimeline.map((g, i) => ({
        label: g.label,
        value: g.settledInferenceSeconds,
        colorClass: ticketColorClass(i),
        valueDisplay: fmtMMSS(g.settledInferenceSeconds),
      })),
    );
  }

  get machineryDonut(): DonutSlice[] {
    let t = this.totals;
    return buildDonut([
      {
        label: 'Validation',
        value: t?.validationSeconds ?? 0,
        colorClass: 'validation',
        valueDisplay: fmtMMSS(t?.validationSeconds),
      },
      {
        label: 'Render gate',
        value: t?.renderGateSeconds ?? 0,
        colorClass: 'render',
        valueDisplay: fmtMMSS(t?.renderGateSeconds),
      },
      {
        label: 'Sync',
        value: t?.syncSeconds ?? 0,
        colorClass: 'sync',
        valueDisplay: fmtMMSS(t?.syncSeconds),
      },
      {
        label: 'Startup',
        value: t?.startupSeconds ?? 0,
        colorClass: 'startup',
        valueDisplay: fmtMMSS(t?.startupSeconds),
      },
    ]);
  }

  /* The compactions tile counts turns, so its donut shows what share of
     turns paid a context-window re-ingest. */
  get compactionDonut(): DonutSlice[] {
    let compacted = this.compactionCount;
    let clean = Math.max(0, this.turns.length - compacted);
    return buildDonut([
      {
        label: 'Compacted',
        value: compacted,
        colorClass: 'compacted',
        valueDisplay: `${compacted}`,
      },
      {
        label: 'Clean',
        value: clean,
        colorClass: 'clean',
        valueDisplay: `${clean}`,
      },
    ]);
  }

  /* --------------------------- Outcome badge --------------------------- */

  get outcomeLabel(): string {
    return capitalise(this.outcome || 'unknown');
  }
  get outcomeBadgeClass(): string {
    switch (this.outcome) {
      case 'completed':
        return 'rt-badge--completed';
      case 'running':
        return 'rt-badge--running';
      case 'stopped':
        return 'rt-badge--stopped';
      case 'failed':
        return 'rt-badge--failed';
      default:
        return '';
    }
  }

  /* --------------------------- Timeline -------------------------------- */

  get timelineTotalWeight(): number {
    let sum = 0;
    for (let t of this.turns) sum += this.turnWeight(t);
    return sum || 1;
  }

  get derivedTurns(): DerivedTurn[] {
    return this.turns.map((turn) => {
      let isLive = turn.durationSeconds == null;
      let weight = this.turnWeight(turn);
      // Threshold: segments < 3% of the timeline print label BELOW.
      // 3% × 900px ≈ 27px — comfortably below the ~44px needed for "MM:SS".
      let ratio = weight / this.timelineTotalWeight;
      let narrow = ratio < 0.03;
      let display = fmtShortMMSS(isLive ? this.turnElapsedSeconds(turn) : turn.durationSeconds);
      return {
        turn,
        colorClass: isLive ? 'live' : turnColorClass(turn.turnType),
        isLive,
        durationSeconds: weight,
        durationDisplay: display,
        narrowLabel: narrow,
      };
    });
  }

  /* Group derivedTurns by ticket, preserving first-appearance order.
     `label` is the grouping key: the board key when the turn carries one,
     the issue title otherwise. Falling back to a single shared literal
     would fold every ticket in the run into one band.

     `weight` is the ticket's whole wall-clock cost — its turns plus the
     validation / render / sync work the loop ran for it — so the timeline
     and the per-ticket subtotals agree. */
  get issueGroupsForTimeline() {
    let byLabel = new Map<
      string,
      {
        label: string;
        issueId: string;
        issueTitle: string;
        turns: DerivedTurn[];
        summary: IssueSummaryField | undefined;
        inferenceSeconds: number;
        /* Settled turns only — comparable with totals.inferenceSeconds,
           which never counts a turn that hasn't closed. */
        settledInferenceSeconds: number;
        liveSeconds: number;
        machinerySeconds: number;
        weight: number;
        hasLive: boolean;
      }
    >();
    for (let dt of this.derivedTurns) {
      let id = dt.turn.issueId ?? '';
      let title = dt.turn.issueTitle ?? '';
      let label = id || title || 'unattributed';
      let g = byLabel.get(label);
      if (!g) {
        let summary = this.issues.find((i) => i.issueId === label);
        let machinery =
          (summary?.validationSeconds ?? 0) +
          (summary?.renderGateSeconds ?? 0) +
          (summary?.syncSeconds ?? 0);
        g = {
          label,
          issueId: id,
          issueTitle: title || label,
          turns: [],
          summary,
          inferenceSeconds: 0,
          settledInferenceSeconds: 0,
          liveSeconds: 0,
          machinerySeconds: machinery,
          weight: machinery,
          hasLive: false,
        };
        byLabel.set(label, g);
      }
      g.turns.push(dt);
      g.inferenceSeconds += dt.durationSeconds;
      g.weight += dt.durationSeconds;
      if (dt.isLive) {
        g.hasLive = true;
        g.liveSeconds += dt.durationSeconds;
      } else {
        g.settledInferenceSeconds += dt.durationSeconds;
      }
    }
    return Array.from(byLabel.values());
  }

  get timelineHint(): string {
    let turns = this.turns.length;
    let tickets = this.ticketSegments.length;
    return `${turns} turn${turns === 1 ? '' : 's'} · ${tickets} ticket${tickets === 1 ? '' : 's'}`;
  }

  /* One timeline segment per ticket. Coloring by ticket rather than by turn
     type is what makes the band readable: turn-type coloring repeated the
     same hue for a ticket's design turn and another ticket's fix turn. */
  get ticketSegments() {
    let groups = this.issueGroupsForTimeline;
    let total = groups.reduce((sum, g) => sum + g.weight, 0) || 1;
    return groups.map((g, i) => {
      let ratio = g.weight / total;
      return {
        label: g.label,
        title: g.issueTitle,
        pill: statusPill(g.summary?.status),
        colorClass: g.hasLive ? 'live' : ticketColorClass(i),
        weight: g.weight,
        durationDisplay: fmtShortMMSS(g.weight),
        narrowLabel: ratio < 0.06,
        hasLive: g.hasLive,
        tooltip: `${g.label} · ${g.issueTitle} · inference ${fmtMMSS(g.inferenceSeconds)} · machinery ${fmtMMSS(g.machinerySeconds)}`,
      };
    });
  }

  /* --------------------------- Axis marks ------------------------------ */

  get axisMarks(): number[] {
    let seconds = this.wallClockDisplaySeconds;
    let mins = Math.max(1, Math.ceil(seconds / 60));
    let step: number;
    if (mins <= 5) step = 1;
    else if (mins <= 15) step = 2;
    else step = 5;
    let marks: number[] = [];
    for (let m = 0; m <= mins; m += step) marks.push(m);
    if (marks[marks.length - 1] !== mins) marks.push(mins);
    return marks;
  }

  /* --------------------------- Event ticks ----------------------------- */

  get eventTicks() {
    let started = this.startedAtMs;
    let wall = this.totals?.wallClockSeconds || this.wallClockDisplaySeconds || 1;
    let ticks = this.events
      .map((ev, idx) => {
        let ts = ev.timestamp as unknown as string | undefined;
        if (!ts || !started) return null;
        let ms = new Date(ts).getTime();
        if (!Number.isFinite(ms)) return null;
        let pct = ((ms - started) / 1000 / wall) * 100;
        if (!Number.isFinite(pct)) return null;
        pct = Math.max(0, Math.min(100, pct));
        return {
          idx,
          leftPct: pct,
          label: (ev.kind ?? '').replace(/-/g, ' '),
          detail: ev.detail ?? '',
          below: false as boolean,
          // Labels at the extremes anchor inward so they aren't clipped.
          edgeClass: pct < 6 ? 'rt-tl-event--start' : pct > 94 ? 'rt-tl-event--end' : '',
        };
      })
      .filter((x): x is NonNullable<typeof x> => x !== null)
      .sort((a, b) => a.leftPct - b.leftPct);

    // Flip the SECOND of any pair that lies within 8% horizontally below
    // the track so the labels don't overlap. Walk pairwise, respecting
    // the previous element's placement.
    let lastAbovePct: number | null = null;
    let lastBelowPct: number | null = null;
    for (let t of ticks) {
      let collidesAbove = lastAbovePct != null && Math.abs(t.leftPct - lastAbovePct) < 8;
      let collidesBelow = lastBelowPct != null && Math.abs(t.leftPct - lastBelowPct) < 8;
      if (collidesAbove && !collidesBelow) {
        t.below = true;
        lastBelowPct = t.leftPct;
      } else {
        t.below = false;
        lastAbovePct = t.leftPct;
      }
    }
    return ticks;
  }

  /* --------------------------- Per-ticket breakdown -------------------- */

  get issueRows() {
    let totalRun = this.totals?.wallClockSeconds || this.wallClockDisplaySeconds || 1;
    return this.issueGroupsForTimeline.map((g, index) => {
      let issueSummary = g.summary;
      let iterationCount = issueSummary?.iterationCount ?? 1;
      let subtotalOutput = 0;
      let subtotalCache = 0;
      let subtotalFresh = 0;
      let inferenceSeconds = 0;
      let hasLive = false;
      for (let dt of g.turns) {
        subtotalOutput += dt.turn.outputTokens ?? 0;
        subtotalCache += dt.turn.cacheReadTokens ?? 0;
        subtotalFresh += dt.turn.freshInputTokens ?? 0;
        if (dt.isLive) hasLive = true;
        else inferenceSeconds += dt.turn.durationSeconds ?? 0;
      }
      let machinerySeconds = g.machinerySeconds;
      let totalSeconds = inferenceSeconds + machinerySeconds;
      let shareText = this.isRunning
        ? hasLive
          ? 'running now'
          : `${fmtPercent(totalSeconds, totalRun)} of run so far`
        : `${fmtPercent(totalSeconds, totalRun)} of run`;

      let title = issueSummary?.title ?? g.issueTitle;
      return {
        issueId: g.issueId,
        title,
        pill: statusPill(issueSummary?.status),
        colorClass: ticketColorClass(index),
        // The board key prefixes the title only when the turns carried one.
        heading: g.issueId && g.issueId !== title ? `${g.issueId} · ${title}` : title,
        // Everything the factory did for this ticket, in order: the agent
        // turns first, then the machinery those turns triggered.
        activityRows: [
          ...g.turns.map((dt) => this.turnRowFor(dt)),
          ...this.mechanicsRowsFor(issueSummary),
        ],
        subtotal: {
          turnCount: g.turns.length,
          durationDisplay: fmtMMSS(totalSeconds),
          outputDisplay: fmtNumber(subtotalOutput),
          cacheDisplay: fmtNumber(subtotalCache),
          hitRateDisplay: fmtHitRate(subtotalFresh, subtotalCache),
        },
        subtotalDurationDisplay: fmtMMSS(totalSeconds),
        splitText: `${fmtMMSS(inferenceSeconds)} inference · ${fmtMMSS(machinerySeconds)} machinery`,
        shareText,
        hasLive,
        iterationCount,
        validationStepsRun: issueSummary?.validationStepsRun ?? 0,
        validationPassed: issueSummary?.validationPassed ?? false,
        ranValidation: (issueSummary?.validationStepsRun ?? 0) > 0,
        emphasiseIterations: iterationCount > 2,
      };
    });
  }

  /* Non-inference work the loop did for one ticket. Rows appear only when
     the phase actually ran, so an empty row never implies zero cost. */
  mechanicsRowsFor(summary: IssueSummaryField | undefined) {
    if (!summary) return [];
    let steps = summary.validationStepsRun ?? 0;
    let syncs = summary.syncCallCount ?? 0;
    let rows: {
      colorClass: string;
      turnLabel: string;
      compacted: boolean;
      modelText: string;
      durationDisplay: string;
      outputDisplay: string;
      cacheDisplay: string;
      hitRateDisplay: string;
      isLive: boolean;
      isMechanics: boolean;
    }[] = [];
    let push = (label: string, detail: string, seconds: number | undefined) => {
      rows.push({
        colorClass: 'machinery',
        turnLabel: label,
        compacted: false,
        modelText: detail,
        durationDisplay: fmtMMSS(seconds),
        outputDisplay: '—',
        cacheDisplay: '—',
        hitRateDisplay: '—',
        isLive: false,
        isMechanics: true,
      });
    };
    if ((summary.validationSeconds ?? 0) > 0 || steps > 0) {
      push(
        'validation',
        `${steps} step${steps === 1 ? '' : 's'} run`,
        summary.validationSeconds,
      );
    }
    if ((summary.renderGateSeconds ?? 0) > 0) {
      push('render gate', 'card screenshots', summary.renderGateSeconds);
    }
    if ((summary.syncSeconds ?? 0) > 0 || syncs > 0) {
      push(
        'workspace sync',
        `${syncs} push${syncs === 1 ? '' : 'es'}`,
        summary.syncSeconds,
      );
    }
    return rows;
  }

  turnRowFor(dt: DerivedTurn) {
    let modelParts: string[] = [];
    if (dt.turn.model) modelParts.push(dt.turn.model as unknown as string);
    if (dt.turn.effort) modelParts.push(dt.turn.effort as unknown as string);
    let modelText = modelParts.join(' · ');
    let turnLabel = dt.isLive
      ? `${dt.turn.turnType} · live`
      : dt.turn.turnType ?? '';
    return {
      colorClass: dt.colorClass,
      turnLabel,
      compacted: !!dt.turn.compacted,
      modelText,
      durationDisplay: dt.isLive
        ? `${fmtMMSS(this.turnElapsedSeconds(dt.turn))} →`
        : fmtMMSS(dt.turn.durationSeconds),
      outputDisplay: dt.isLive ? '—' : fmtNumber(dt.turn.outputTokens),
      cacheDisplay: dt.isLive ? '—' : fmtNumber(dt.turn.cacheReadTokens),
      hitRateDisplay: dt.isLive
        ? '—'
        : fmtHitRate(dt.turn.freshInputTokens ?? 0, dt.turn.cacheReadTokens ?? 0),
      isLive: dt.isLive,
      isMechanics: false,
    };
  }

  get grandTotal() {
    let output = 0;
    let cache = 0;
    let fresh = 0;
    let inference = 0;
    let liveTurns = 0;
    for (let t of this.turns) {
      output += t.outputTokens ?? 0;
      cache += t.cacheReadTokens ?? 0;
      fresh += t.freshInputTokens ?? 0;
      if (t.durationSeconds != null) inference += t.durationSeconds;
      else liveTurns += 1;
    }
    // Machinery only counts once here: the per-ticket rows already show it,
    // and the grand row must add up to the same number they do.
    let machinery = this.issueGroupsForTimeline.reduce(
      (sum, g) => sum + g.machinerySeconds,
      0,
    );
    return {
      turnCount: this.turns.length,
      issueCount: this.issues.length,
      durationDisplay: fmtMMSS(inference + machinery),
      outputDisplay: fmtNumber(output),
      cacheDisplay: fmtNumber(cache),
      hitRateDisplay: fmtHitRate(fresh, cache),
      labelText: this.isRunning ? 'All issues · run so far' : 'All issues · run total',
      metaText: `${this.turns.length} turn${this.turns.length === 1 ? '' : 's'} · ${this.issues.length} issue${this.issues.length === 1 ? '' : 's'}`,
      hasLive: liveTurns > 0,
    };
  }

  /* --------------------------- Token economics ------------------------- */

  get tokenRows() {
    return this.turns
      .filter((t) => t.durationSeconds != null) // live turn omitted until settled
      .map((t) => {
        return {
          colorClass: turnColorClass(t.turnType),
          turnLabel: `${t.issueId ?? ''} · ${t.turnType ?? ''}`,
          compacted: !!t.compacted,
          freshDisplay: fmtNumber(t.freshInputTokens),
          cacheDisplay: fmtNumber(t.cacheReadTokens),
          hitRateDisplay: fmtHitRate(t.freshInputTokens ?? 0, t.cacheReadTokens ?? 0),
        };
      });
  }

  get tokenGrandTotal() {
    let fresh = 0;
    let cache = 0;
    for (let t of this.turns) {
      if (t.durationSeconds == null) continue;
      fresh += t.freshInputTokens ?? 0;
      cache += t.cacheReadTokens ?? 0;
    }
    return {
      freshDisplay: fmtNumber(fresh),
      cacheDisplay: fmtNumber(cache),
      hitRateDisplay: fmtHitRate(fresh, cache),
    };
  }

  get tokenCallout() {
    // Find the settled turn with the highest fresh-input count.
    let settled = this.turns.filter((t) => t.durationSeconds != null);
    if (settled.length === 0) {
      return {
        leadStrong: 'No settled turns yet.',
        leadRest: ' The callout will rewrite itself around the highest fresh-input turn once the run has more data.',
      };
    }
    let ranked = settled.slice().sort(
      (a, b) => (b.freshInputTokens ?? 0) - (a.freshInputTokens ?? 0),
    );
    let top = ranked[0];
    let topFresh = top.freshInputTokens ?? 0;

    // Compute the ratio against a "normal" turn (the median of the remaining
    // settled turns, floored at 1 to avoid division by zero).
    let others = ranked.slice(1);
    let ratioText = '';
    if (others.length > 0) {
      let median =
        others
          .map((t) => t.freshInputTokens ?? 0)
          .sort((a, b) => a - b)[Math.floor(others.length / 2)] || 1;
      let ratio = topFresh / Math.max(1, median);
      if (ratio >= 10) ratioText = ` — ~${Math.round(ratio)}× a normal turn`;
    }

    let isAnomaly = top.compacted || topFresh > 5000;
    if (isAnomaly) {
      let leadStrong = `${top.issueId ?? ''} ${top.turnType ?? ''} ${top.compacted ? 're-ingested' : 'consumed'} ${fmtNumber(topFresh)} fresh tokens${top.compacted ? ' after compaction' : ''}${ratioText}.`;
      return {
        leadStrong,
        leadRest: ' Cached input is nearly free; fresh input and output are the spend.',
      };
    }
    return {
      leadStrong: `${top.issueId ?? ''} ${top.turnType ?? ''} carried the most fresh input at ${fmtNumber(topFresh)} tokens.`,
      leadRest: ' Cached input is nearly free; fresh input and output are the spend.',
    };
  }

  get hasTokenRows(): boolean {
    return this.tokenRows.length > 0;
  }

  /* --------------------------- Phase subtotals ------------------------- */

  get visiblePhases() {
    let wall = this.totals?.wallClockSeconds || this.wallClockDisplaySeconds || 1;
    return this.phases
      .filter((p) => (p.wallClockSeconds ?? 0) > 0)
      .map((p) => ({
        name: p.name ?? '',
        label: capitalise(p.name),
        colorClass: phaseColorClass(p.name),
        seconds: p.wallClockSeconds ?? 0,
        secondsDisplay: fmtMMSS(p.wallClockSeconds),
        pctDisplay: fmtPercent(p.wallClockSeconds ?? 0, wall),
        widthPct: Math.min(100, ((p.wallClockSeconds ?? 0) / wall) * 100),
      }));
  }

  get phaseSectionHint(): string {
    return this.isRunning
      ? `${this.visiblePhases.length} phase${this.visiblePhases.length === 1 ? '' : 's'} so far`
      : `${this.visiblePhases.length} phase${this.visiblePhases.length === 1 ? '' : 's'} ran`;
  }

  /* --------------------------- Skills in context ----------------------- */

  /* One block per lifecycle phase, listing the skills the agent's context
     carried during that phase. The writer already orders rows by phase then
     by weight, so grouping preserves its order. */
  get skillPhaseGroups() {
    let byPhase = new Map<
      string,
      {
        phase: string;
        label: string;
        colorClass: string;
        characters: number;
        rows: {
          name: string;
          turnTypes: string;
          turnCountDisplay: string;
          charactersDisplay: string;
          referencesDisplay: string;
          deliveryLabel: string;
          deliveryClass: string;
        }[];
      }
    >();
    for (let s of this.skills) {
      let phase = s.phase ?? '';
      let g = byPhase.get(phase);
      if (!g) {
        g = {
          phase,
          label: capitalise(phase),
          colorClass: phaseColorClass(phase),
          characters: 0,
          rows: [],
        };
        byPhase.set(phase, g);
      }
      let turns = s.turnCount ?? 0;
      let refs = s.referenceCount ?? 0;
      g.characters += s.characters ?? 0;
      g.rows.push({
        name: s.name ?? '',
        turnTypes: s.turnTypes ?? '',
        turnCountDisplay: `${turns} turn${turns === 1 ? '' : 's'}`,
        charactersDisplay: fmtNumber(s.characters),
        referencesDisplay: refs === 0 ? '—' : fmtNumber(refs),
        deliveryLabel: deliveryLabel(s.delivery),
        deliveryClass: `rt-tag--${s.delivery ?? 'injected'}`,
      });
    }
    return Array.from(byPhase.values()).map((g) => ({
      ...g,
      subtotalDisplay: `${g.rows.length} skill${g.rows.length === 1 ? '' : 's'} · ${fmtNumber(g.characters)} chars`,
    }));
  }

  get hasSkillRows(): boolean {
    return this.skills.length > 0;
  }

  get skillSectionHint(): string {
    let names = new Set(this.skills.map((s) => s.name ?? ''));
    return `${names.size} distinct skill${names.size === 1 ? '' : 's'} across ${this.skillPhaseGroups.length} phase${this.skillPhaseGroups.length === 1 ? '' : 's'}`;
  }

/* =============================================================================
 * Embedded component — header + KPI row + phase bars only.
 * ============================================================================= */

class RunTelemetryEmbedded extends Component<typeof RunTelemetry> {
  @tracked nowMs = Date.now();
  #ticker: ReturnType<typeof setInterval> | undefined;

  constructor(owner: unknown, args: any) {
    super(owner as any, args);
    this.#ticker = setInterval(() => {
      this.nowMs = Date.now();
    }, 1000);
    registerDestructor(this, () => {
      if (this.#ticker) clearInterval(this.#ticker);
    });
  }

  get config(): RunConfigField | undefined {
    return this.args.model?.config as RunConfigField | undefined;
  }
  get totals(): TotalsField | undefined {
    return this.args.model?.totals as TotalsField | undefined;
  }
  get turns(): TurnField[] {
    return (this.args.model?.turns as TurnField[] | undefined) ?? [];
  }
  get phases(): PhaseField[] {
    return (this.args.model?.phases as PhaseField[] | undefined) ?? [];
  }
  get issues(): IssueSummaryField[] {
    return (this.args.model?.issues as IssueSummaryField[] | undefined) ?? [];
  }
  get outcome(): string {
    return this.config?.outcome ?? '';
  }
  get isRunning(): boolean {
    return this.outcome === 'running';
  }

  get startedAtMs(): number {
    let s = this.config?.startedAt as unknown as string | undefined;
    if (!s) return 0;
    let d = new Date(s).getTime();
    return Number.isFinite(d) ? d : 0;
  }
  get elapsedSeconds(): number {
    if (!this.startedAtMs) return 0;
    return Math.max(0, Math.floor((this.nowMs - this.startedAtMs) / 1000));
  }
  get wallClockDisplaySeconds(): number {
    if (this.isRunning) {
      return Math.max(this.elapsedSeconds, this.totals?.wallClockSeconds ?? 0);
    }
    return this.totals?.wallClockSeconds ?? 0;
  }
  get wallClockDisplay(): string {
    return fmtMMSS(this.wallClockDisplaySeconds);
  }

  get inferenceDisplay(): string {
    return fmtMMSS(this.totals?.inferenceSeconds);
  }
  get inferencePercent(): string {
    let wall = this.wallClockDisplaySeconds;
    let inf = this.totals?.inferenceSeconds ?? 0;
    if (!wall) return '—';
    return `${Math.round((inf / wall) * 100)}%`;
  }
  get inferenceFootText(): string {
    return `${this.inferencePercent} of wall clock · ${this.turns.length} turn${this.turns.length === 1 ? '' : 's'}`;
  }

  get machinerySeconds(): number {
    let t = this.totals;
    if (!t) return 0;
    return (
      (t.validationSeconds ?? 0) +
      (t.renderGateSeconds ?? 0) +
      (t.syncSeconds ?? 0) +
      (t.startupSeconds ?? 0)
    );
  }
  get machineryDisplay(): string {
    return fmtMMSS(this.machinerySeconds);
  }
  get machineryFootText(): string {
    let t = this.totals;
    if (!t) return '';
    let syncCalls = t.syncCallCount ?? 0;
    return `validation ${fmtMMSS(t.validationSeconds)} · render ${fmtMMSS(t.renderGateSeconds)} · sync ${fmtMMSS(t.syncSeconds)} (${syncCalls})`;
  }

  get compactionCount(): number {
    return this.turns.reduce((n, t) => n + (t.compacted ? 1 : 0), 0);
  }
  get compactionFootText(): string {
    let n = this.compactionCount;
    if (n === 0) return 'no context-window overflows';
    return `${n} × ~17k fresh tokens re-ingested`;
  }

  get outcomeLabel(): string {
    return capitalise(this.outcome || 'unknown');
  }
  get outcomeBadgeClass(): string {
    switch (this.outcome) {
      case 'completed':
        return 'rt-badge--completed';
      case 'running':
        return 'rt-badge--running';
      case 'stopped':
        return 'rt-badge--stopped';
      case 'failed':
        return 'rt-badge--failed';
      default:
        return '';
    }
  }

  get runIndex(): string {
    let url = (this.config?.targetRealmUrl as unknown as string) ?? '';
    let m = /v(\d+)/i.exec(url);
    return m ? m[1] : '1';
  }
  get eyebrowText(): string {
    return `Run ${this.runIndex} · ${capitalise(this.config?.targetPhase)}`;
  }
  get headerTitle(): string {
    let m = this.args.model as any;
    if (m?.cardInfo?.name) return m.cardInfo.name as string;
    if (m?.title) return m.title as string;
    return 'Untitled Run Telemetry';
  }
  get subtitleParts() {
    let issues = this.issues.length;
    let turns = this.turns.length;
    if (this.isRunning) {
      return {
        before: `${issues} issue${issues === 1 ? '' : 's'} started, ${turns} agent turn${turns === 1 ? '' : 's'} so far, `,
        strong: this.wallClockDisplay,
        after: ' elapsed on the wall clock.',
      };
    }
    // Only tickets that actually ran a validation pipeline have a verdict;
    // reading "every issue completed" off a default would be a fabrication.
    let validated = this.issues.filter((i) => (i.validationStepsRun ?? 0) > 0);
    let after: string;
    if (validated.length === 0) {
      after = ' — no validation pipeline ran.';
    } else if (validated.every((i) => i.validationPassed)) {
      after = ` with all ${validated.length} validated ticket${validated.length === 1 ? '' : 's'} passing.`;
    } else {
      after = ' with mixed validation outcomes.';
    }
    return {
      before: `${issues} issue${issues === 1 ? '' : 's'}, ${turns} agent turn${turns === 1 ? '' : 's'}, done in `,
      strong: fmtMMSS(this.totals?.wallClockSeconds),
      after,
    };
  }

  get visiblePhases() {
    let wall = this.totals?.wallClockSeconds || this.wallClockDisplaySeconds || 1;
    return this.phases
      .filter((p) => (p.wallClockSeconds ?? 0) > 0)
      .map((p) => ({
        name: p.name ?? '',
        label: capitalise(p.name),
        colorClass: phaseColorClass(p.name),
        secondsDisplay: fmtMMSS(p.wallClockSeconds),
        pctDisplay: fmtPercent(p.wallClockSeconds ?? 0, wall),
        widthPct: Math.min(100, ((p.wallClockSeconds ?? 0) / wall) * 100),
      }));
  }

  <template>
    <div class='rt-embed'>
      <div class='rt-header'>
        <div class='rt-eyebrow'>{{this.eyebrowText}}</div>
        <h2 class='rt-title'>{{this.headerTitle}}</h2>
        <div class='rt-sub'>
          {{this.subtitleParts.before}}<strong>{{this.subtitleParts.strong}}</strong>{{this.subtitleParts.after}}
        </div>
      </div>

      <div class='rt-kpi-row'>

          <div class='rt-kpi-value'>{{this.wallClockDisplay}}</div>

            <span class='rt-badge {{this.outcomeBadgeClass}}'>{{this.outcomeLabel}}</span>
          </div>
        </div>
        <div class='rt-kpi'>
          <div class='rt-eyebrow'>Model inference</div>
          <div class='rt-kpi-value rt-kpi-value--lifted'>{{this.inferenceDisplay}}</div>
          <div class='rt-kpi-foot'>{{this.inferenceFootText}}</div>
        </div>
        <div class='rt-kpi'>
          <div class='rt-eyebrow'>Everything else</div>
          <div class='rt-kpi-value'>{{this.machineryDisplay}}</div>
          <div class='rt-kpi-foot'>{{this.machineryFootText}}</div>

        <div class='rt-kpi'>

          <div class='rt-kpi-value'>{{this.compactionCount}}</div>
          <div class='rt-kpi-foot'>{{this.compactionFootText}}</div>
        </div>
      </div>

      <div class='rt-phases'>
        {{#each this.visiblePhases as |p|}}
          <div class='rt-phase-row'>
            <div class='rt-phase-label'>{{p.label}}</div>
            <div class='rt-phase-track'>
              <div
                class='rt-phase-fill rt-phase-fill--{{p.colorClass}}'
                style={{htmlSafeWidth p.widthPct}}
              ></div>
            </div>
            <div class='rt-phase-val'>{{p.secondsDisplay}}</div>
            <div class='rt-phase-pct'>{{p.pctDisplay}}</div>
          </div>
        {{/each}}
      </div>
    </div>

    <style scoped>

        --rt-surface: #ffffff;

    let d = new Date(s).getTime();
    return Number.isFinite(d) ? d : 0;
  }
  get elapsedSeconds(): number {
    if (!this.startedAtMs) return 0;
    return Math.max(0, Math.floor((this.nowMs - this.startedAtMs) / 1000));
  }
  get wallClockDisplay(): string {
    if (this.isRunning) {
      return fmtMMSS(
        Math.max(this.elapsedSeconds, this.totals?.wallClockSeconds ?? 0),
      );
    }
    return fmtMMSS(this.totals?.wallClockSeconds);
  }
  get inferenceDisplay(): string {
    return fmtMMSS(this.totals?.inferenceSeconds);
  }
  get runIndex(): string {
    let url = (this.config?.targetRealmUrl as unknown as string) ?? '';
    let m = /v(\d+)/i.exec(url);
    return m ? m[1] : '1';
  }
  get eyebrowText(): string {
    let base = `Run ${this.runIndex} · ${capitalise(this.config?.targetPhase)}`;
    return this.isRunning ? `${base} · Live` : base;
  }
  get headerTitle(): string {
    let m = this.args.model as any;
    if (m?.cardInfo?.name) return m.cardInfo.name as string;
    if (m?.title) return m.title as string;
    return 'Untitled Run Telemetry';
  }
  get outcomeLabel(): string {
    return capitalise(this.outcome || 'unknown');
  }
  get outcomeBadgeClass(): string {
    switch (this.outcome) {
      case 'completed':
        return 'rt-badge--completed';
      case 'running':
        return 'rt-badge--running';
      case 'stopped':
        return 'rt-badge--stopped';
      case 'failed':
        return 'rt-badge--failed';
      default:
        return '';
    }
  }
  get footMeta(): string {
    return `${this.turns.length} turn${this.turns.length === 1 ? '' : 's'} · ${this.issues.length} issue${this.issues.length === 1 ? '' : 's'}`;
  }

  <template>
    <div class='fit'>
      <div class='fit-eyebrow {{if this.isRunning "fit-eyebrow--live"}}'>
        {{this.eyebrowText}}
      </div>
      <div class='fit-title'>{{this.headerTitle}}</div>
      <div class='fit-nums'>
        <div class='fit-nums-cell'>
          <div class='fit-nums-eyebrow'>Total</div>
          <div class='fit-nums-value'>{{this.wallClockDisplay}}</div>
        </div>
        <div class='fit-nums-cell'>
          <div class='fit-nums-eyebrow'>Inference</div>
          <div class='fit-nums-value fit-nums-value--lifted'>{{this.inferenceDisplay}}</div>
        </div>
      </div>
      <div class='fit-foot'>
        <span class='fit-foot-meta'>{{this.footMeta}}</span>

      </div>
    </div>

    <style scoped>

/* =============================================================================
 * Style helpers — return SafeString values so Glimmer accepts them in
 * inline `style={{ }}` bindings without triggering the HTML-escape guard.
 * ============================================================================= */

function htmlSafeFlex(weight: number) {
  return htmlSafe(`flex: ${Math.max(0, weight)};`);
}
function htmlSafeWidth(pct: number) {
  let p = Math.max(0, Math.min(100, pct));
  return htmlSafe(`width: ${p}%;`);
}
function htmlSafeLeft(pct: number) {
  let p = Math.max(0, Math.min(100, pct));
  return htmlSafe(`left: ${p}%;`);
}
function isNotOne(n: number): boolean {
  return n !== 1;
}
